SQL · SQLITE · DATABASE DESIGN 01
SQL / SQLITE RELATIONAL DATA 
SELECT
    c.Country,
    COUNT(DISTINCT c.CustomerId)
        AS "Number of Customers",
    COUNT(DISTINCT i.InvoiceId)
        AS "Number of Invoices",
    ROUND(SUM(i.Total), 2)
        AS "Total Purchase Value"

FROM customers c

LEFT JOIN invoices i
    ON c.CustomerId = i.CustomerId

GROUP BY c.Country

ORDER BY
    "Total Purchase Value" DESC;
